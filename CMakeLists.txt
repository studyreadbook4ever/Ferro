cmake_minimum_required(VERSION 3.20)
project(Ferro VERSION 2.2.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Concurrent Test)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PORTAUDIO REQUIRED IMPORTED_TARGET portaudio-2.0)
pkg_check_modules(SNDFILE REQUIRED IMPORTED_TARGET sndfile)
add_executable(ferro src/main.cpp src/project.cpp src/dsp.cpp src/audio.cpp src/supervisor.cpp src/window.cpp src/window.h src/timeline.cpp src/timeline.h src/supervisor.h tests/project_tests.cpp tests/dsp_tests.cpp tests/audio_tests.cpp tests/supervisor_tests.cpp)
target_link_libraries(ferro PRIVATE Qt6::Core Qt6::Widgets Qt6::Concurrent Qt6::Test PkgConfig::PORTAUDIO PkgConfig::SNDFILE)
target_include_directories(ferro PRIVATE src)
target_compile_options(ferro PRIVATE -Wall -Wextra -Wpedantic)
option(FERRO_SANITIZE "Enable address/undefined behavior sanitizers" OFF)
if(FERRO_SANITIZE)
  target_compile_options(ferro PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(ferro PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME engine COMMAND ferro --self-test)
set_tests_properties(engine PROPERTIES TIMEOUT 180)
add_test(NAME gui COMMAND ferro --smoke-gui)
set_tests_properties(gui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 45)
install(TARGETS ferro RUNTIME DESTINATION bin)
