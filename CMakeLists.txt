cmake_minimum_required(VERSION 3.20)
project(Ferro VERSION 2.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Concurrent Test Network)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PORTAUDIO REQUIRED IMPORTED_TARGET portaudio-2.0)
pkg_check_modules(SNDFILE REQUIRED IMPORTED_TARGET sndfile)
add_executable(ferro src/main.cpp src/project.cpp src/agent.cpp src/agent_ipc.cpp src/dsp.cpp src/audio.cpp src/supervisor.cpp src/window.cpp src/window.h src/timeline.cpp src/timeline.h src/supervisor.h assets/showcase.qrc tests/project_tests.cpp tests/agent_tests.cpp tests/dsp_tests.cpp tests/audio_tests.cpp tests/supervisor_tests.cpp)
target_link_libraries(ferro PRIVATE Qt6::Core Qt6::Widgets Qt6::Concurrent Qt6::Test Qt6::Network PkgConfig::PORTAUDIO PkgConfig::SNDFILE)
target_include_directories(ferro PRIVATE src)
target_compile_options(ferro PRIVATE -Wall -Wextra -Wpedantic)
option(FERRO_SANITIZE "Enable address/undefined behavior sanitizers" OFF)
if(FERRO_SANITIZE)
  target_compile_options(ferro PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(ferro PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME engine COMMAND ferro --self-test)
set_tests_properties(engine PROPERTIES TIMEOUT 180)
add_test(NAME gui COMMAND ferro --smoke-gui)
set_tests_properties(gui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 45)
find_package(Python3 QUIET COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME agent_cli COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/agent_cli_tests.py" "$<TARGET_FILE:ferro>" "${CMAKE_CURRENT_SOURCE_DIR}")
  set_tests_properties(agent_cli PROPERTIES TIMEOUT 180)
endif()
install(TARGETS ferro RUNTIME DESTINATION bin)
install(PROGRAMS ferro-agent DESTINATION bin)
install(FILES docs/AGENT_CLI.md DESTINATION share/ferro/docs)
install(FILES examples/Neon-Signal-2011.ferro examples/agent-four-bars.json DESTINATION share/ferro/examples)
